# Projekt: LAN Network

## Opis

Building with two floors, four rooms on each. Internet connected to bridge by fiber on first floor. Then first of four Deco x20, then 5 ports switch. 
From switch: 
1st wire cat 6 to 2nd switch which connects TV, PS5 and laptop. 
2nd wire cat 6 to sons room. 
3rd wire cat 6 to 2nd floor

### scan_network.sh

```
#!/bin/bash
# ============================================
# Skaner sieci domowej - Archer AC1200
# Uruchom na laptopie: bash scan_network.sh
# ============================================

echo "=========================================="
echo "  SKANER SIECI DOMOWEJ"
echo "=========================================="
echo ""

# 1. Informacje o interfejsie sieciowym
echo "=== TWÓJ INTERFEJS SIECIOWY ==="
if command -v ip &> /dev/null; then
    ip addr show | grep -E "inet |link/ether" | head -10
    echo ""
    echo "--- Brama domyślna ---"
    ip route show default
elif command -v ifconfig &> /dev/null; then
    ifconfig | grep -E "inet |ether" | head -10
    echo ""
    echo "--- Brama domyślna ---"
    netstat -rn | head -5
fi
echo ""

# 2. Skan ARP - szybki przegląd urządzeń
echo "=== TABLICA ARP (znane urządzenia) ==="
if command -v arp &> /dev/null; then
    arp -a 2>/dev/null
elif command -v ip &> /dev/null; then
    ip neigh show
fi
echo ""

# 3. Skan nmap (jeśli zainstalowany)
if command -v nmap &> /dev/null; then
    echo "=== NMAP - PEŁNY SKAN SIECI ==="
    # Wykryj podsieć
    SUBNET=$(ip route show | grep -v default | grep "src" | head -1 | awk '{print $1}')
    if [ -z "$SUBNET" ]; then
        SUBNET="192.168.0.0/24"
    fi
    echo "Skanuję podsieć: $SUBNET"
    echo ""
    
    echo "--- Wykrywanie hostów (ping scan) ---"
    nmap -sn $SUBNET 2>/dev/null
    echo ""
    
    echo "--- Szczegółowy skan routera ---"
    GATEWAY=$(ip route show default 2>/dev/null | awk '{print $3}' | head -1)
    if [ -z "$GATEWAY" ]; then
        GATEWAY="192.168.0.1"
    fi
    echo "Skanuję bramę: $GATEWAY"
    nmap -sV -O $GATEWAY 2>/dev/null
    echo ""
else
    echo "=== NMAP NIE ZAINSTALOWANY ==="
    echo "Zainstaluj nmap dla pełnego skanu:"
    echo "  Ubuntu/Debian: sudo apt install nmap"
    echo "  macOS:         brew install nmap"
    echo "  Windows:       choco install nmap"
    echo ""
    
    # Alternatywny szybki skan pingiem
    echo "=== ALTERNATYWNY SKAN PINGIEM ==="
    GATEWAY=$(ip route show default 2>/dev/null | awk '{print $3}' | head -1)
    if [ -z "$GATEWAY" ]; then
        GATEWAY="192.168.0.1"
    fi
    BASE=$(echo $GATEWAY | cut -d. -f1-3)
    echo "Skanuję ${BASE}.0/24..."
    for i in $(seq 1 254); do
        ping -c 1 -W 1 ${BASE}.${i} &>/dev/null && echo "  AKTYWNY: ${BASE}.${i}" &
    done
    wait
    echo ""
fi

# 4. Informacje o Wi-Fi
echo "=== INFORMACJE O WI-FI ==="
if command -v iwconfig &> /dev/null; then
    iwconfig 2>/dev/null | grep -v "no wireless"
elif command -v nmcli &> /dev/null; then
    nmcli dev wifi show-password 2>/dev/null
    nmcli connection show --active 2>/dev/null
fi
echo ""

# 5. DNS
echo "=== KONFIGURACJA DNS ==="
cat /etc/resolv.conf 2>/dev/null | grep -v "^#"
echo ""

# 6. Sprawdź panel routera
echo "=== TEST PANELU ROUTERA ==="
GATEWAY=$(ip route show default 2>/dev/null | awk '{print $3}' | head -1)
if [ -z "$GATEWAY" ]; then
    GATEWAY="192.168.0.1"
fi
echo -n "Panel na http://${GATEWAY}: "
if command -v curl &> /dev/null; then
    HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" --connect-timeout 3 http://${GATEWAY}/)
    echo "HTTP $HTTP_CODE"
elif command -v wget &> /dev/null; then
    wget -q --spider --timeout=3 http://${GATEWAY}/ && echo "DOSTĘPNY" || echo "NIEDOSTĘPNY"
fi
echo ""

echo "=========================================="
echo "  SKAN ZAKOŃCZONY"
echo "  Wklej te wyniki do Claude!"
echo "=========================================="

```
